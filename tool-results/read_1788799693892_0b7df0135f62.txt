   429→
   430→            {authed ? (
   431→              <DropdownMenu>
   432→                <DropdownMenuTrigger asChild>
   433→                  <button
   434→                    aria-label="Meniu cont"
   435→                    className="flex h-9 items-center gap-2 rounded-full bg-zinc-900 pl-1.5 pr-3 ring-1 ring-zinc-800 hover:ring-zinc-600 transition"
   436→                  >
   437→                    {session?.user?.image ? (
   438→                       
   439→                      <img src={session.user.image} alt="Avatar" className="h-6 w-6 rounded-full" referrerPolicy="no-referrer" />
   440→                    ) : (
   441→                      <span className="flex h-6 w-6 items-center justify-center rounded-full bg-red-600 text-xs font-black text-white">
   442→                        {(session?.user?.name || "U").charAt(0).toUpperCase()}
   443→                      </span>
   444→                    )}
   445→                    <span className="hidden max-w-[120px] truncate text-xs font-semibold text-zinc-200 sm:block">
   446→                      {session?.user?.name}
   447→                    </span>
   448→                    <User className="h-3.5 w-3.5 text-zinc-500" />
   449→                  </button>
   450→                </DropdownMenuTrigger>

   577→      />
   578→
