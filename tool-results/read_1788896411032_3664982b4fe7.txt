   399→                } else if (e.key === "Enter" && sugIdx >= 0) {
   400→                  e.preventDefault();
   401→                  pickSuggestion(suggestions[sugIdx]);
   402→                } else if (e.key === "Escape") {
   403→                  setShowSug(false);
   404→                }
   405→              }}
   406→              placeholder="Caută filme, seriale, anime, muzică..."
   407→              className="h-9 border-zinc-800 bg-zinc-900/80 pl-8 text-sm text-zinc-200 placeholder:text-zinc-500 focus-visible:ring-red-600/50"

   409→              aria-expanded={showSug && suggestions.length > 0}

   413→              <div className="absolute left-0 right-0 top-full z-30 mt-1 overflow-hidden rounded-xl border border-zinc-800 bg-zinc-950/95 shadow-2xl shadow-black/60 backdrop-blur">

   417→                {suggestions.map((s, i) => (

   425→                    }`}
   426→                  >
   427→                    <Search className="h-3.5 w-3.5 shrink-0 text-zinc-600" />
