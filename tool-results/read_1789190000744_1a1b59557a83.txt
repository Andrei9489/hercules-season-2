     1→"use client";
     2→
     3→// ============================================================
     4→// LibraryAddDialog — STUDIOUL DE ÎNCĂRCARE UNIVERSAL (Faza 8+9)
     5→// Utilizatorul încarcă conținutul REAL al platformei:
     6→//   • link direct: MP4 / WebM / HLS (.m3u8) / DASH (.mpd)
     7→//   • platforme: YouTube, OK.ru, Vimeo, TikTok, Dailymotion,
     8→//     Rumble, Twitch, Facebook, VK, Streamable, Drive, Bilibili...
     9→//   • cod embed: <iframe> / <script> / <object> / <video>
    10→//   • orice sursă necunoscută → iframe generic cu fallback extern
    11→//   • PLAYLIST M3U/M3U8 (Faza 9): paste conținut sau URL → canale
    12→//     TV/radio cu tvg-logo, group-title multi-categorii, calitate,
    13→//     [Geo-blocked]/[Not 24/7] → bulk idempotent în Neon
    14→// MOD BULK: un element pe linie (max 50 / batch) — metadatele
    15→// comune (tip / brand / țară / an / genuri) se aplică tuturor.
    16→// Metadatele se salvează în NEON — nimic local.
    17→// ============================================================
    18→
    19→import { useEffect, useMemo, useState } from "react";
    20→import { Dialog, DialogContent, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    21→import { Input } from "@/components/ui/input";
    22→import { toast } from "@/hooks/use-toast";
    23→import { resolveSource, titleFromUrl } from "@/lib/source-resolver";
    24→import { parseM3U } from "@/lib/m3u-parser";
    25→import type { LibraryItem } from "./types";
    26→import { api } from "./api";
    27→
    28→type Props = {
    29→  open: boolean;
    30→  onClose: () => void;
    31→  onAdded: (item?: LibraryItem) => void;
    32→};
    33→
    34→const TYPES = [
    35→  ["video", "Video"], ["movie", "Film"], ["series", "Serial"], ["anime", "Anime"],
    36→  ["cartoon", "Desene animate"], ["documentary", "Documentar"], ["music", "Muzică"],
    37→  ["sport", "Sport"], ["gaming", "Gaming"], ["news", "Știri"], ["radio", "Radio"],
    38→  ["showbiz", "Show-biz"], ["telenovela", "Telenovelă"],
    39→];
    40→
    41→const BRANDS = ["", "marvel", "dc", "disney", "pixar", "cartoon-network", "jetix", "fox-kids", "boomerang", "minimax", "ghibli"];
    42→
    43→/** Împarte textarea-ul pe linii — fiecare linie = o sursă (URL sau cod embed). */
    44→function splitSources(text: string): string[] {
    45→  return text
    46→    .split(/\n+/)
    47→    .map((l) => l.trim())
    48→    .filter(Boolean);
    49→}
    50→
    51→type M3UImportResult = {
    52→  ok: boolean;
    53→  playlistName: string;
    54→  parsed: number;
    55→  parsedRadio: number;
    56→  added: number;
    57→  duplicates: number;
    58→  skipped: number;
    59→  errors: string[];
    60→  topGroups: { group: string; n: number }[];
    61→};
    62→
    63→// FAZA 16 — rezultatul gărzii de duplicate la salvare
    64→type DupInfo = { title: string; reason: string; id?: number };
    65→type AddResult = { added: number; duplicates: DupInfo[]; failed: string[]; items: LibraryItem[] };
    66→
    67→const REASON_LABEL: Record<string, string> = {
    68→  external_id: "același ID de sursă",
    69→  source: "același link / stream",
    70→  title: "titlu identic (același tip)",
    71→};
    72→
    73→export function LibraryAddDialog({ open, onClose, onAdded }: Props) {
    74→  const [mode, setMode] = useState<"sources" | "m3u">("sources");
    75→
    76→  // ---- mod SURSE (Faza 8) ----
    77→  const [input, setInput] = useState("");
    78→  const [title, setTitle] = useState("");
    79→  const [posterUrl, setPosterUrl] = useState("");
    80→  const [description, setDescription] = useState("");
    81→  const [year, setYear] = useState("");
    82→  const [genres, setGenres] = useState("");
    83→  const [type, setType] = useState("video");
    84→  const [brand, setBrand] = useState("");
    85→  const [country, setCountry] = useState("");
    86→  const [saving, setSaving] = useState(false);
    87→
    88→  // ---- Faza 10: ACCES SECURIZAT (token/HMAC/JWT) ----
    89→  const [secOn, setSecOn] = useState(false);
    90→  const [secType, setSecType] = useState("query");
    91→  const [secParam, setSecParam] = useState("token");
    92→  const [secSecret, setSecSecret] = useState("");
    93→  const [secTtl, setSecTtl] = useState("300");
    94→
    95→  // ---- mod PLAYLIST M3U (Faza 9) ----
    96→  const [m3uText, setM3uText] = useState("");
    97→  const [m3uUrl, setM3uUrl] = useState("");
    98→  const [importing, setImporting] = useState(false);
    99→
   100→  // ---- FAZA 16: gardă duplicate LIVE (alertă înainte de salvare) ----
   101→  const [dupWarn, setDupWarn] = useState<{ title: string; reason: string; id: number } | null>(null);
   102→  const [forceAdd, setForceAdd] = useState(false);
   103→
   104→  const lines = useMemo(() => splitSources(input), [input]);
   105→
   106→  const detected = useMemo(() => {
   107→    if (lines.length === 0) return null;
   108→    try {
   109→      return resolveSource(lines[0], { parent: typeof window !== "undefined" ? window.location.hostname : "" });
   110→    } catch {
   111→      return null;
   112→    }
   113→  }, [lines]);
   114→
   115→  const detectedCount = useMemo(() => {
   116→    let ok = 0;
   117→    for (const l of lines) {
   118→      try { if (resolveSource(l, {})) ok++; } catch { /* nerecunoscut */ }
   119→    }
   120→    return ok;
   121→  }, [lines]);
   122→
   123→  // preview parsare M3U client-side (parserul e server-safe = merge și în browser)
   124→  const m3uPreview = useMemo(() => {
   125→    if (mode !== "m3u" || !m3uText.trim()) return null;
   126→    try {
   127→      const r = parseM3U(m3uText);
   128→      const groups = new Map<string, number>();
   129→      for (const ch of r.channels) for (const g of ch.groups) groups.set(g, (groups.get(g) || 0) + 1);
   130→      return {
   131→        channels: r.channels.length,
   132→        radio: r.channels.filter((c) => c.isRadio).length,
   133→        playlistName: r.playlistName,
   134→        errors: r.errors,
   135→        topGroups: [...groups.entries()].sort((a, b) => b[1] - a[1]).slice(0, 12),
   136→        sample: r.channels.slice(0, 4).map((c) => c.name),
   137→      };
   138→    } catch {
   139→      return null;
   140→    }
   141→  }, [mode, m3uText]);
   142→
   143→  // titlu automat din URL (doar în mod singular)
   144→  useEffect(() => {
   145→    if (mode !== "sources" || lines.length !== 1 || !lines[0] || title) return;
   146→    const d = detected;
   147→    if (!d) return;
   148→    const ref = d.kind === "html" ? "" : (d as { src?: string; url?: string }).src || (d as { url?: string }).url || "";
   149→    if (ref.startsWith("http")) setTitle(titleFromUrl(ref));
   150→  }, [mode, lines, detected, title]);
   151→
   152→  // FAZA 16 — verificare duplicat LIVE (debounce 450ms): dacă titlul + tipul
   153→  // sau sursa există deja în bibliotecă → ALERTĂ vizibilă înainte de salvare.
   154→  useEffect(() => {
   155→    if (mode !== "sources" || lines.length !== 1 || !lines[0]) {
   156→      setDupWarn(null);
   157→      return;
   158→    }
   159→    let cancelled = false;
   160→    const t = setTimeout(async () => {
   161→      try {
   162→        const d = detected;
   163→        const ref = d && d.kind !== "html" ? ((d as { src?: string; url?: string }).src || (d as { url?: string }).url || "") : "";
   164→        const effTitle = title.trim() || (ref ? titleFromUrl(ref) : lines[0].slice(0, 60));
   165→        const r = await api.managePost<{ duplicate: boolean; matches: { id: number; title: string; reason: string }[] }>({
   166→          action: "check",
   167→          title: effTitle,
   168→          contentType: type,
   169→          sourceUrl: ref || undefined,
   170→        });
   171→        if (cancelled) return;
   172→        setDupWarn(r.duplicate && r.matches[0] ? { title: r.matches[0].title, reason: r.matches[0].reason, id: r.matches[0].id } : null);
   173→      } catch {
   174→        if (!cancelled) setDupWarn(null);
   175→      }
   176→    }, 450);
   177→    return () => { cancelled = true; clearTimeout(t); };
   178→  }, [mode, lines, detected, title, type]);
   179→
   180→  const reset = () => {
   181→    setInput(""); setTitle(""); setPosterUrl(""); setDescription("");
   182→    setYear(""); setGenres(""); setType("video"); setBrand(""); setCountry("");
   183→    setSecOn(false); setSecSecret(""); setSecParam("token"); setSecType("query"); setSecTtl("300");
   184→    setM3uText(""); setM3uUrl("");
   185→    setDupWarn(null); setForceAdd(false);
   186→  };
   187→
   188→  const save = async () => {
   189→    if (lines.length === 0) {
   190→      toast({ title: "Introdu cel puțin un link sau cod embed", variant: "destructive" });
   191→      return;
   192→    }
   193→    setSaving(true);
   194→    try {
   195→      const bulk = lines.length > 1;
   196→      // Faza 10 — semnarea se aplică doar în mod singular, pe streamuri directe
   197→      const direct = detected && ["video", "hls", "dash", "ts"].includes(detected.kind);
   198→      const signing = secOn && !bulk && direct && secSecret.trim().length >= 4
   199→        ? { type: secType, param: secParam.trim() || "token", secret: secSecret.trim(), ttlSec: Number(secTtl) || 300 }
   200→        : undefined;
   201→      const r = await api.libraryPost<AddResult>({
   202→        action: "add",
   203→        items: bulk ? lines : undefined,
   204→        input: bulk ? undefined : lines[0],
   205→        title: title.trim() || undefined,
   206→        posterUrl: posterUrl.trim() || undefined,
   207→        description: description.trim() || undefined,
   208→        year: year.trim() || undefined,
   209→        genres: genres.trim() || undefined,
   210→        contentType: type,
   211→        brand: brand || undefined,
   212→        country: country.trim() || undefined,
   213→        signing,
   214→        force: forceAdd || undefined,
   215→      });
   216→      const parts = [`${r.added} salvat(e) în Neon ✅`];
   217→      if (r.duplicates?.length) {
   218→        const d0 = r.duplicates[0];
   219→        parts.push(`⚠️ ${r.duplicates.length} BLOCAT(E) — există deja (${REASON_LABEL[d0.reason] || d0.reason}): „${d0.title.slice(0, 48)}”`);
   220→      }
   221→      if (r.failed?.length) parts.push(`${r.failed.length} eșuat(e)`);
   222→      toast({
   223→        title: parts.join(" • "),
   224→        description: bulk
   225→          ? `${lines.length} surse procesate`
   226→          : r.duplicates?.length
   227→            ? `⚠️ Acest conținut există deja în bibliotecă (id ${r.duplicates[0].id ?? "?"}) — nu a fost adăugat de 2 ori.`
   228→            : `${r.items?.[0]?.title || lines[0].slice(0, 50)}`,
   229→        variant: r.duplicates?.length && r.added === 0 ? "destructive" : undefined,
   230→      });
   231→      onAdded(r.items?.[0]);
   232→      reset();
   233→      onClose();
   234→    } catch {
   235→      toast({ title: "Eroare la salvare în Neon", variant: "destructive" });
   236→    } finally {
   237→      setSaving(false);
   238→    }
   239→  };
   240→
   241→  const importM3U = async () => {
   242→    if (!m3uText.trim() && !m3uUrl.trim()) {
   243→      toast({ title: "Lipește conținutul M3U sau introdu un URL de playlist", variant: "destructive" });
   244→      return;
   245→    }
   246→    setImporting(true);
   247→    try {
   248→      const r = await api.libraryPost<M3UImportResult>({ action: "m3u", content: m3uText || undefined, url: m3uUrl || undefined });
   249→      const parts = [`${r.added} canale importate în Neon ✅`];
   250→      if (r.duplicates) parts.push(`${r.duplicates} duplicate`);
   251→      if (r.skipped) parts.push(`${r.skipped} ignorate`);
   252→      toast({
   253→        title: parts.join(" • "),
   254→        description: `„${r.playlistName}" — ${r.parsed} canale${r.parsedRadio ? ` (${r.parsedRadio} radio)` : ""}${r.topGroups.length ? ` • grupuri: ${r.topGroups.slice(0, 3).map((g) => g.group).join(", ")}` : ""}`,
   255→      });
   256→      onAdded();
   257→      reset();
   258→      onClose();
   259→    } catch {
   260→      toast({ title: "Eroare la importul M3U — verifică formatul/URL-ul", variant: "destructive" });
   261→    } finally {
   262→      setImporting(false);
   263→    }
   264→  };
   265→
   266→  return (
   267→    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
   268→      <DialogContent aria-describedby={undefined} className="max-h-[92vh] max-w-lg overflow-y-auto border-zinc-800 bg-zinc-950 text-zinc-100">
   269→        <DialogHeader>
   270→          <DialogTitle className="text-lg font-black">Studiul de încărcare universal</DialogTitle>
   271→        </DialogHeader>
   272→
   273→        {/* comutator mod: Surse | Playlist M3U */}
   274→        <div className="flex rounded-lg bg-zinc-900 p-1 ring-1 ring-zinc-800" role="tablist" aria-label="Mod încărcare">
   275→          <button
   276→            role="tab"
   277→            aria-selected={mode === "sources"}
   278→            onClick={() => setMode("sources")}
   279→            className={`flex-1 rounded-md px-3 py-1.5 text-xs font-bold transition ${mode === "sources" ? "bg-red-600 text-white" : "text-zinc-400 hover:text-zinc-200"}`}
   280→          >
   281→            Linkuri & Embed
   282→          </button>
   283→          <button
   284→            role="tab"
   285→            aria-selected={mode === "m3u"}
   286→            onClick={() => setMode("m3u")}
   287→            className={`flex-1 rounded-md px-3 py-1.5 text-xs font-bold transition ${mode === "m3u" ? "bg-red-600 text-white" : "text-zinc-400 hover:text-zinc-200"}`}
   288→          >
   289→            Playlist M3U / IPTV
   290→          </button>
   291→        </div>
   292→
   293→        {mode === "sources" ? (
   294→          <>
   295→            <p className="-mt-1 text-xs text-zinc-500">
   296→              Lipește <b className="text-zinc-300">un element pe linie</b> — link de redare
   297→              (YouTube, OK.ru, Vimeo, TikTok, Dailymotion, Rumble, Twitch, MP4, HLS, DASH...) sau
   298→              <b className="text-zinc-300"> cod embed</b> (iframe / script / HTML). Titlul și miniatura
   299→              reale se extrag automat unde e posibil.
   300→            </p>
   301→
   302→            <textarea
   303→              value={input}
   304→              onChange={(e) => setInput(e.target.value)}
   305→              placeholder={"https://www.youtube.com/watch?v=...\nhttps://ok.ru/video/12345\nhttps://vimeo.com/76979871\n<iframe src=\"https://www.dailymotion.com/embed/video/x8...\"></iframe>\nhttps://example.com/film.mp4\nhttps://stream.tv/live/index.m3u8"}
   306→              rows={5}
   307→              className="w-full resize-y rounded-lg border border-zinc-800 bg-zinc-900 px-3 py-2 font-mono text-xs text-zinc-200 placeholder:text-zinc-600 focus:outline-none focus:ring-2 focus:ring-red-600/50"
   308→            />
   309→
   310→            {input.trim() && (
   311→              <div className={`flex items-center gap-2 rounded-lg px-3 py-2 text-xs ring-1 ${
   312→                detectedCount > 0 ? "bg-emerald-500/10 text-emerald-300 ring-emerald-600/30" : "bg-red-500/10 text-red-300 ring-red-600/30"
   313→              }`}>
   314→                {detectedCount > 0 ? (
   315→                  <>
   316→                    ✓ {detectedCount}/{lines.length} sursă/e detectată/e
   317→                    {lines.length === 1 && detected && (
   318→                      <> • <b>{detected.providerLabel}</b> • redare: {detected.kind === "iframe" ? "iframe embed" : detected.kind === "video" ? "player nativ" : detected.kind === "hls" ? "HLS adaptiv" : detected.kind === "dash" ? "DASH adaptiv" : detected.kind === "ts" ? "MPEG-TS (mpegts.js)" : detected.kind === "unplayable" ? `protocol ${detected.protocol.toUpperCase()} — player extern (VLC)` : detected.kind === "html" ? "embed sandboxat" : "iframe generic"}</>
   319→                    )}
   320→                    {lines.length > 1 && <span className="text-emerald-400/70">— mod BULK</span>}
   321→                  </>
   322→                ) : (
   323→                  <>✗ Nicio sursă recunoscută — dar poți încerca oricum redarea generică</>
   324→                )}
   325→              </div>
   326→            )}
   327→
   328→            {lines.length <= 1 && dupWarn && (
   329→              <div className="rounded-lg bg-amber-500/10 px-3 py-2 text-xs text-amber-200 ring-1 ring-amber-600/40" data-duplicate-alert>
   330→                <b>⚠️ Există deja în bibliotecă:</b> „{dupWarn.title.slice(0, 60)}” (id {dupWarn.id}) — {REASON_LABEL[dupWarn.reason] || dupWarn.reason}.
   331→                Salvarea va fi <b>BLOCATĂ</b> — nu se pot încărca două conținuturi la fel.
   332→                {dupWarn.reason === "title" && (
   333→                  <label className="mt-1.5 flex items-center gap-2 font-medium text-amber-100">
   334→                    <input
   335→                      type="checkbox"
   336→                      checked={forceAdd}
   337→                      onChange={(e) => setForceAdd(e.target.checked)}
   338→                      className="h-3.5 w-3.5 accent-amber-500"
   339→                    />
   340→                    Este alt conținut (titlu similar, sursă diferită) — adaugă oricum
   341→                  </label>
   342→                )}
   343→              </div>
   344→            )}
   345→
   346→            {lines.length <= 1 && (
   347→              <Input
   348→                value={title}
   349→                onChange={(e) => setTitle(e.target.value)}
   350→                placeholder="Titlu (auto din link / oEmbed, opțional editabil)"
   351→                className="border-zinc-800 bg-zinc-900 text-sm"
   352→              />
   353→            )}
   354→
   355→            <Input
   356→              value={posterUrl}
   357→              onChange={(e) => setPosterUrl(e.target.value)}
   358→              placeholder="URL poster (opțional — altfel miniatură automată)"
   359→              className="border-zinc-800 bg-zinc-900 text-sm"
   360→            />
   361→
   362→            <textarea
   363→              value={description}
   364→              onChange={(e) => setDescription(e.target.value)}
   365→              placeholder="Descriere (opțional)"
   366→              rows={2}
   367→              className="w-full resize-none rounded-lg border border-zinc-800 bg-zinc-900 px-3 py-2 text-sm text-zinc-200 placeholder:text-zinc-600 focus:outline-none focus:ring-2 focus:ring-red-600/50"
   368→            />
   369→
   370→            <div className="grid grid-cols-2 gap-2">
   371→              <select
   372→                value={type}
   373→                onChange={(e) => setType(e.target.value)}
   374→                aria-label="Tip conținut"
   375→                className="h-9 rounded-lg border border-zinc-800 bg-zinc-900 px-2 text-sm text-zinc-200"
   376→              >
   377→                {TYPES.map(([v, l]) => <option key={v} value={v}>{l}</option>)}
   378→              </select>
   379→              <select
   380→                value={brand}
   381→                onChange={(e) => setBrand(e.target.value)}
   382→                aria-label="Brand / univers"
   383→                className="h-9 rounded-lg border border-zinc-800 bg-zinc-900 px-2 text-sm text-zinc-200"
   384→              >
   385→                {BRANDS.map((b) => <option key={b || "none"} value={b}>{b ? b.replace(/-/g, " ") : "— fără brand —"}</option>)}
   386→              </select>
   387→            </div>
   388→
   389→            <div className="grid grid-cols-3 gap-2">
   390→              <Input
   391→                value={year}
   392→                onChange={(e) => setYear(e.target.value.replace(/\D/g, "").slice(0, 4))}
   393→                placeholder="An (ex: 2024)"
   394→                className="border-zinc-800 bg-zinc-900 text-sm"
   395→              />
   396→              <Input
   397→                value={country}
   398→                onChange={(e) => setCountry(e.target.value)}
   399→                placeholder="Țară"
   400→                className="border-zinc-800 bg-zinc-900 text-sm"
   401→              />
   402→              <Input
   403→                value={genres}
   404→                onChange={(e) => setGenres(e.target.value)}
   405→                placeholder="Genuri (virgulă)"
   406→                className="border-zinc-800 bg-zinc-900 text-sm"
   407→              />
   408→            </div>
   409→
   410→            {/* ---------- FAZA 10: ACCES SECURIZAT (token/HMAC/JWT) ---------- */}
   411→            {lines.length <= 1 && (
   412→              <details className="rounded-lg border border-zinc-800 bg-zinc-900/60" open={secOn}>
   413→                <summary className="flex cursor-pointer list-none items-center justify-between px-3 py-2 text-xs font-bold text-zinc-300">
   414→                  <span>🔐 Acces securizat (token / HMAC / JWT)</span>
   415→                  <button
   416→                    type="button"
   417→                    aria-label="Comută semnarea"
   418→                    onClick={(e) => { e.preventDefault(); setSecOn((v) => !v); }}
   419→                    className={`rounded-full px-2 py-0.5 text-[10px] font-bold ${secOn ? "bg-emerald-600 text-white" : "bg-zinc-800 text-zinc-400"}`}
   420→                  >
   421→                    {secOn ? "ACTIV" : "OFF"}
   422→                  </button>
   423→                </summary>
   424→                {secOn && (
   425→                  <div className="space-y-2 px-3 pb-3">
   426→                    <p className="text-[11px] leading-relaxed text-zinc-500">
   427→                      Pentru streamuri directe protejate (MP4/HLS/DASH/MPEG-TS). Secretul se salvează
   428→                      în Neon și NU ajunge niciodată în browser — la redare, serverul semnează URL-ul.
   429→                    </p>
   430→                    <div className="grid grid-cols-2 gap-2">
   431→                      <select
   432→                        value={secType}
   433→                        onChange={(e) => setSecType(e.target.value)}
   434→                        aria-label="Schema de semnare"
   435→                        className="h-9 rounded-lg border border-zinc-800 bg-zinc-950 px-2 text-xs text-zinc-200"
   436→                      >
   437→                        <option value="query">Query token (?token=secret)</option>
   438→                        <option value="hmac-md5">HMAC-MD5 (Wowza/Flussonic)</option>
