   147→

   149→

   160→    if (!authed) {

   167→      const r = await api.userPost<{ active?: boolean }>({
   168→        action: "toggle", kind,
   169→        media: {
   170→          mediaId: i.id, mediaType: i.mediaType, title: i.title, poster: i.poster,
   171→          backdrop: i.backdrop, year: i.year, rating: i.rating, source: i.source || "tmdb",
   172→        },
   173→      });
   174→      setter((prev) => {
   175→        const next = new Set(prev);
   176→        if (r.active) next.add(active); else next.delete(active);
   177→        return next;
   178→      });
   179→      setRefreshKey((k) => k + 1);
   180→      toast({
   181→        title: r.active ? "Adăugat ✅" : "Eliminat",
   182→        description: `${i.title} — ${kind === "watchlist" ? "watchlist" : "favorite"}`,
   183→      });
   184→    } catch {
   185→      toast({ title: "Eroare la salvare", variant: "destructive" });
   186→    }
   187→  }, [authed]);
   188→
   189→  const openDetail = useCallback((i: MediaItem) => {
   190→    setDetailItem(i); setDetailOpen(true);
   191→  }, []);
   192→
   193→  const openPlayer = useCallback((i: MediaItem) => {
   194→    setPlayerItem(i); setPlayerOpen(true);
   195→  }, []);
   196→
   197→  const navigate = useCallback((v: string) => {
   198→    setView(v as ViewKey);
   199→    setSidebarOpen(false);
   200→    window.scrollTo({ top: 0 });
   201→  }, []);
   202→

   204→  const navigateExtra = useCallback((item: ExtraItem) => {
   205→    if (item.view === "universuri") setUniversuriTab(item.param);
   206→    else if (item.view === "showbiz") setShowbizTab(item.param);
   207→    else if (item.view === "copii") setKidsBrand(item.param);
   208→    else if (item.view === "stiri") setNewsContinent(item.param);
   209→    setView(item.view);
   210→    setSidebarOpen(false);
   211→    window.scrollTo({ top: 0 });
   212→  }, []);
   213→
   214→  // autocompletare live: debounce 180ms → /api/search?mode=suggest (Neon)
   215→  // q gol → trending din search_stats (upsert_search_stat)
   216→  useEffect(() => {
   217→    const t = setTimeout(() => {
   218→      sugAbort.current?.abort();
   219→      const ac = new AbortController();
   220→      sugAbort.current = ac;
   221→      fetch(`/api/search?mode=suggest&q=${encodeURIComponent(searchInput.trim())}`, {
   222→        signal: ac.signal,
   223→        cache: "no-store",
   224→      })
   225→        .then((r) => (r.ok ? r.json() : { suggestions: [] }))
   226→        .then((d: { suggestions?: string[] }) => {
   227→          if (ac.signal.aborted) return;
   228→          setSuggestions((d.suggestions || []).slice(0, 7));
   229→          setSugIdx(-1);
   230→        })
   231→        .catch(() => {});
   232→    }, 180);
   233→    return () => clearTimeout(t);
   234→  }, [searchInput]);
   235→
   236→  const pickSuggestion = useCallback(
   237→    (s: string) => {
   238→      setSearchInput(s);
   239→      setSearchQuery(s);
   240→      setShowSug(false);
   241→      navigate("search");
   242→    },
   243→    [navigate]
   244→  );
   245→
   246→  const submitSearch = (e: React.FormEvent) => {
   247→    e.preventDefault();
   248→    if (searchInput.trim()) {
   249→      setSearchQuery(searchInput.trim());
   250→      setShowSug(false);
   251→      navigate("search");
   252→    }
   253→  };
   254→
   255→  const addProfile = async () => {
   256→    if (!newProfile.name.trim()) return;
   257→    try {
   258→      const p = await api.userPost<Profile>({
   259→        action: "add", kind: "profiles",
   260→        profile: { name: newProfile.name.trim(), avatar: newProfile.avatar },
   261→      });
   262→      setProfiles((prev) => [...prev, p]);
   263→      setActiveProfile(p);
   264→      setProfileDialogOpen(false);
   265→      setNewProfile({ name: "", avatar: "🎬" });
   266→      toast({ title: "Profil creat ✅", description: p.name });
   267→    } catch {
   268→      toast({ title: "Eroare", description: "Nu am putut crea profilul.", variant: "destructive" });
   269→    }
   270→  };
   271→
   272→  const groups = Array.from(new Set(NAV.map((n) => n.group)));
   273→
   274→  const sidebarContent = (
   275→    <>
   276→      <div className="mb-6 flex items-center gap-2 px-4">
   277→        <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-red-600 text-lg font-black shadow-lg shadow-red-600/30">S</div>
   278→        <div>
   279→          <p className="text-base font-black leading-none tracking-tight">Stream<span className="text-red-500">Verse</span></p>
   280→          <p className="mt-0.5 text-[10px] text-zinc-500">196 țări • 6 continente</p>
   281→        </div>
   282→      </div>
   283→
   284→      <nav className="flex-1 space-y-5 overflow-y-auto px-2 pb-4">
   285→        {groups.map((g) => (
   286→          <div key={g}>
   287→            <p className="mb-1 px-3 text-[10px] font-bold uppercase tracking-widest text-zinc-600">{g}</p>
   288→            {NAV.filter((n) => n.group === g).map((n) => (
   289→              <button
   290→                key={n.key}
   291→                onClick={() => navigate(n.key)}
   292→                className={`mb-0.5 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition ${

   294→                    ? "bg-red-600/15 text-red-400 ring-1 ring-red-600/30"

   309→              const active = view === n.view &&
   310→                (n.view === "universuri" ? universuriTab === n.param
   311→                  : n.view === "showbiz" ? showbizTab === n.param
   312→                  : n.view === "copii" ? kidsBrand === n.param

   314→              return (
   315→                <button
   316→                  key={n.label}
   317→                  onClick={() => navigateExtra(n)}
   318→                  title={`Deschide ${n.label}`}
   319→                  className={`mb-0.5 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition ${
   320→                    active
   321→                      ? "bg-red-600/15 text-red-400 ring-1 ring-red-600/30"
   322→                      : "text-zinc-400 hover:bg-zinc-900 hover:text-zinc-100"
   323→                  }`}
   324→                >
   325→                  <span className="w-4 shrink-0 text-center text-base leading-none">{n.icon}</span>
   326→                  <span className="flex-1 truncate">{n.label}</span>
   327→                </button>
   328→              );
   329→            })}
   330→          </div>
   331→        ))}
   332→      </nav>
   333→
   334→      <div className="border-t border-zinc-900 p-4">
   335→        {authed ? (
   336→          <div className="text-xs text-zinc-500">
   337→            <p className="truncate font-semibold text-zinc-300">{session?.user?.name}</p>

   341→          <p className="text-xs text-zinc-600">Conectează-te pentru a salva favoritele și progresul.</p>

   343→      </div>
   344→    </>
   345→  );
   346→
   347→  return (
   348→    <div className="min-h-screen flex flex-col bg-[#0a0a0f]">

   353→
   354→      {/* SIDEBAR mobile */}

   356→        <div className="fixed inset-0 z-50 lg:hidden">
   357→          <div className="absolute inset-0 bg-black/70 backdrop-blur-sm" onClick={() => setSidebarOpen(false)} />

   359→            <button
   360→              aria-label="Închide meniul"
   361→              onClick={() => setSidebarOpen(false)}

   384→            <Input
   385→              value={searchInput}
   386→              onChange={(e) => { setSearchInput(e.target.value); setShowSug(true); }}
   387→              onFocus={() => setShowSug(true)}
   388→              onBlur={() => setTimeout(() => setShowSug(false), 120)}
   389→              onKeyDown={(e) => {
   390→                if (!showSug || suggestions.length === 0) return;
   391→                if (e.key === "ArrowDown") {
   392→                  e.preventDefault();
   393→                  setSugIdx((i) => (i + 1) % suggestions.length);
   394→                } else if (e.key === "ArrowUp") {
   395→                  e.preventDefault();
   396→                  setSugIdx((i) => (i <= 0 ? suggestions.length - 1 : i - 1));
   397→                } else if (e.key === "Enter" && sugIdx >= 0) {
   398→                  e.preventDefault();
   399→                  pickSuggestion(suggestions[sugIdx]);
   400→                } else if (e.key === "Escape") {
   401→                  setShowSug(false);
   402→                }
   403→              }}
   404→              placeholder="Caută filme, seriale, anime, muzică..."
   405→              className="h-9 border-zinc-800 bg-zinc-900/80 pl-8 text-sm text-zinc-200 placeholder:text-zinc-500 focus-visible:ring-red-600/50"
   406→              aria-autocomplete="list"
   407→              aria-expanded={showSug && suggestions.length > 0}
   408→              role="combobox"
   409→            />
   410→            {showSug && suggestions.length > 0 && (
   411→              <div className="absolute left-0 right-0 top-full z-30 mt-1 overflow-hidden rounded-xl border border-zinc-800 bg-zinc-950/95 shadow-2xl shadow-black/60 backdrop-blur">
   412→                <p className="px-3 pb-1 pt-2 text-[10px] font-bold uppercase tracking-widest text-zinc-600">
   413→                  {searchInput.trim() ? "🧠 Sugestii din biblioteca Neon" : "🔥 Căutări populare acum"}
   414→                </p>
   415→                {suggestions.map((s, i) => (
   416→                  <button

   418→                    key={`${s}:${i}`}
   419→                    onMouseDown={(e) => { e.preventDefault(); pickSuggestion(s); }}
   420→                    onMouseEnter={() => setSugIdx(i)}
   421→                    className={`flex w-full items-center gap-2 px-3 py-2 text-left text-sm transition ${
   422→                      i === sugIdx ? "bg-red-600/20 text-red-300" : "text-zinc-300"
   423→                    }`}
   424→                  >
   425→                    <Search className="h-3.5 w-3.5 shrink-0 text-zinc-600" />
   426→                    <span className="truncate">{s}</span>
   427→                  </button>
   428→                ))}
   429→              </div>
   430→            )}
   431→          </form>
   432→
   433→          <div className="ml-auto flex items-center gap-2">
