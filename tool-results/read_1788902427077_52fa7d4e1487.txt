   175→      toast({
   176→        title: r.active ? "Adăugat ✅" : "Eliminat",

   219→      })

   259→      setProfileDialogOpen(false);
   260→      setNewProfile({ name: "", avatar: "🎬" });

   263→      toast({ title: "Eroare", description: "Nu am putut crea profilul.", variant: "destructive" });

   269→  const sidebarContent = (

   282→            <p className="mb-1 px-3 text-[10px] font-bold uppercase tracking-widest text-zinc-600">{g}</p>

   393→              onKeyDown={(e) => {

   396→                  e.preventDefault();

   401→                } else if (e.key === "Enter" && sugIdx >= 0) {
   402→                  e.preventDefault();
   403→                  pickSuggestion(suggestions[sugIdx]);
   404→                } else if (e.key === "Escape") {
   405→                  setShowSug(false);
   406→                }
   407→              }}
   408→              placeholder="Caută filme, seriale, anime, muzică..."
   409→              className="h-9 border-zinc-800 bg-zinc-900/80 pl-8 text-sm text-zinc-200 placeholder:text-zinc-500 focus-visible:ring-red-600/50"

   411→              aria-expanded={showSug && suggestions.length > 0}
   412→              role="combobox"
   413→            />

   419→                {suggestions.map((s, i) => (
   420→                  <button
