   159→    if (!authed) {

   167→        action: "toggle", kind,
   168→        media: {
   169→          mediaId: i.id, mediaType: i.mediaType, title: i.title, poster: i.poster,
   170→          backdrop: i.backdrop, year: i.year, rating: i.rating, source: i.source || "tmdb",
   171→        },
   172→      });
   173→      setter((prev) => {
   174→        const next = new Set(prev);
   175→        if (r.active) next.add(active); else next.delete(active);
   176→        return next;
   177→      });
   178→      setRefreshKey((k) => k + 1);
   179→      toast({
   180→        title: r.active ? "Adăugat ✅" : "Eliminat",
   181→        description: `${i.title} — ${kind === "watchlist" ? "watchlist" : "favorite"}`,
   182→      });
   183→    } catch {
   184→      toast({ title: "Eroare la salvare", variant: "destructive" });
   185→    }
   186→  }, [authed]);
   187→
   188→  const openDetail = useCallback((i: MediaItem) => {
   189→    setDetailItem(i); setDetailOpen(true);
   190→  }, []);
   191→
   192→  const openPlayer = useCallback((i: MediaItem) => {
   193→    setPlayerItem(i); setPlayerOpen(true);
   194→  }, []);
   195→
   196→  const navigate = useCallback((v: string) => {
   197→    setView(v as ViewKey);
   198→    setSidebarOpen(false);
   199→    window.scrollTo({ top: 0 });
   200→  }, []);
   201→

   203→  const navigateExtra = useCallback((item: ExtraItem) => {
   204→    if (item.view === "universuri") setUniversuriTab(item.param);
   205→    else if (item.view === "showbiz") setShowbizTab(item.param);
   206→    else if (item.view === "copii") setKidsBrand(item.param);
   207→    else if (item.view === "stiri") setNewsContinent(item.param);
   208→    setView(item.view);
   209→    setSidebarOpen(false);
   210→    window.scrollTo({ top: 0 });
   211→  }, []);
   212→
   213→  // autocompletare live: debounce 180ms → /api/search?mode=suggest (Neon)
   214→  // q gol → trending din search_stats (upsert_search_stat)
   215→  useEffect(() => {
   216→    const t = setTimeout(() => {
   217→      sugAbort.current?.abort();
   218→      const ac = new AbortController();
   219→      sugAbort.current = ac;
   220→      fetch(`/api/search?mode=suggest&q=${encodeURIComponent(searchInput.trim())}`, {
   221→        signal: ac.signal,
   222→        cache: "no-store",
   223→      })
   224→        .then((r) => (r.ok ? r.json() : { suggestions: [] }))
   225→        .then((d: { suggestions?: string[] }) => {
   226→          if (ac.signal.aborted) return;
   227→          setSuggestions((d.suggestions || []).slice(0, 7));
   228→          setSugIdx(-1);
   229→        })
   230→        .catch(() => {});
   231→    }, 180);
   232→    return () => clearTimeout(t);
   233→  }, [searchInput]);
   234→
   235→  const pickSuggestion = useCallback(
   236→    (s: string) => {
   237→      setSearchInput(s);
   238→      setSearchQuery(s);
   239→      setShowSug(false);
   240→      navigate("search");
   241→    },
   242→    [navigate]
   243→  );
   244→
   245→  const submitSearch = (e: React.FormEvent) => {
   246→    e.preventDefault();

   248→      setSearchQuery(searchInput.trim());
   249→      setShowSug(false);
   250→      navigate("search");
   251→    }
   252→  };
   253→
   254→  const addProfile = async () => {
   255→    if (!newProfile.name.trim()) return;
   256→    try {
   257→      const p = await api.userPost<Profile>({
   258→        action: "add", kind: "profiles",
   259→        profile: { name: newProfile.name.trim(), avatar: newProfile.avatar },
   260→      });
   261→      setProfiles((prev) => [...prev, p]);
   262→      setActiveProfile(p);
   263→      setProfileDialogOpen(false);
   264→      setNewProfile({ name: "", avatar: "🎬" });
   265→      toast({ title: "Profil creat ✅", description: p.name });
   266→    } catch {
   267→      toast({ title: "Eroare", description: "Nu am putut crea profilul.", variant: "destructive" });
   268→    }
   269→  };
   270→
   271→  const groups = Array.from(new Set(NAV.map((n) => n.group)));
   272→
   273→  const sidebarContent = (
   274→    <>
   275→      <div className="mb-6 flex items-center gap-2 px-4">
   276→        <div className="flex h-9 w-9 items-center justify-center rounded-xl bg-red-600 text-lg font-black shadow-lg shadow-red-600/30">S</div>
   277→        <div>
   278→          <p className="text-base font-black leading-none tracking-tight">Stream<span className="text-red-500">Verse</span></p>
   279→          <p className="mt-0.5 text-[10px] text-zinc-500">196 țări • 6 continente</p>
   280→        </div>
   281→      </div>
   282→
   283→      <nav className="flex-1 space-y-5 overflow-y-auto px-2 pb-4">
   284→        {groups.map((g) => (
   285→          <div key={g}>
   286→            <p className="mb-1 px-3 text-[10px] font-bold uppercase tracking-widest text-zinc-600">{g}</p>
   287→            {NAV.filter((n) => n.group === g).map((n) => (
   288→              <button
   289→                key={n.key}
   290→                onClick={() => navigate(n.key)}
   291→                className={`mb-0.5 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition ${

   293→                    ? "bg-red-600/15 text-red-400 ring-1 ring-red-600/30"

   297→                <n.icon className="h-4 w-4 shrink-0" />

   309→                (n.view === "universuri" ? universuriTab === n.param

   311→                  : n.view === "copii" ? kidsBrand === n.param

   313→              return (
   314→                <button
   315→                  key={n.label}
   316→                  onClick={() => navigateExtra(n)}
   317→                  title={`Deschide ${n.label}`}
   318→                  className={`mb-0.5 flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium transition ${
   319→                    active
   320→                      ? "bg-red-600/15 text-red-400 ring-1 ring-red-600/30"
   321→                      : "text-zinc-400 hover:bg-zinc-900 hover:text-zinc-100"
   322→                  }`}
   323→                >
   324→                  <span className="w-4 shrink-0 text-center text-base leading-none">{n.icon}</span>
   325→                  <span className="flex-1 truncate">{n.label}</span>
   326→                </button>
   327→              );
   328→            })}
   329→          </div>
   330→        ))}
   331→      </nav>
   332→
   333→      <div className="border-t border-zinc-900 p-4">
   334→        {authed ? (
   335→          <div className="text-xs text-zinc-500">
   336→            <p className="truncate font-semibold text-zinc-300">{session?.user?.name}</p>

   340→          <p className="text-xs text-zinc-600">Conectează-te pentru a salva favoritele și progresul.</p>

   342→      </div>
   343→    </>

   345→
   346→  return (

   352→
   353→      {/* SIDEBAR mobile */}

   356→          <div className="absolute inset-0 bg-black/70 backdrop-blur-sm" onClick={() => setSidebarOpen(false)} />

   358→            <button
   359→              aria-label="Închide meniul"

   383→            <Input
   384→              value={searchInput}
   385→              onChange={(e) => { setSearchInput(e.target.value); setShowSug(true); }}
   386→              onFocus={() => setShowSug(true)}
   387→              onBlur={() => setTimeout(() => setShowSug(false), 120)}
   388→              onKeyDown={(e) => {
   389→                if (!showSug || suggestions.length === 0) return;
   390→                if (e.key === "ArrowDown") {
   391→                  e.preventDefault();
   392→                  setSugIdx((i) => (i + 1) % suggestions.length);
   393→                } else if (e.key === "ArrowUp") {
   394→                  e.preventDefault();
   395→                  setSugIdx((i) => (i <= 0 ? suggestions.length - 1 : i - 1));
   396→                } else if (e.key === "Enter" && sugIdx >= 0) {
   397→                  e.preventDefault();
   398→                  pickSuggestion(suggestions[sugIdx]);
   399→                } else if (e.key === "Escape") {
   400→                  setShowSug(false);
   401→                }
   402→              }}
   403→              placeholder="Caută filme, seriale, anime, muzică..."
   404→              className="h-9 border-zinc-800 bg-zinc-900/80 pl-8 text-sm text-zinc-200 placeholder:text-zinc-500 focus-visible:ring-red-600/50"
   405→              aria-autocomplete="list"
   406→              aria-expanded={showSug && suggestions.length > 0}
   407→              role="combobox"
   408→            />
   409→            {showSug && suggestions.length > 0 && (
   410→              <div className="absolute left-0 right-0 top-full z-30 mt-1 overflow-hidden rounded-xl border border-zinc-800 bg-zinc-950/95 shadow-2xl shadow-black/60 backdrop-blur">
   411→                <p className="px-3 pb-1 pt-2 text-[10px] font-bold uppercase tracking-widest text-zinc-600">
   412→                  {searchInput.trim() ? "🧠 Sugestii din biblioteca Neon" : "🔥 Căutări populare acum"}
   413→                </p>
   414→                {suggestions.map((s, i) => (
   415→                  <button
   416→                    type="button"
   417→                    key={`${s}:${i}`}
   418→                    onMouseDown={(e) => { e.preventDefault(); pickSuggestion(s); }}
   419→                    onMouseEnter={() => setSugIdx(i)}
   420→                    className={`flex w-full items-center gap-2 px-3 py-2 text-left text-sm transition ${
   421→                      i === sugIdx ? "bg-red-600/20 text-red-300" : "text-zinc-300"
   422→                    }`}
   423→                  >
   424→                    <Search className="h-3.5 w-3.5 shrink-0 text-zinc-600" />
   425→                    <span className="truncate">{s}</span>
   426→                  </button>
   427→                ))}
   428→              </div>
   429→            )}
   430→          </form>
   431→
   432→          <div className="ml-auto flex items-center gap-2">
   433→            {/* profiluri */}
